cmake_minimum_required(VERSION 3.25)
project(StreamlineProbe LANGUAGES CXX)
add_executable(StreamlineProbe probe.cpp)
target_compile_features(StreamlineProbe PRIVATE cxx_std_20)
target_compile_definitions(StreamlineProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(StreamlineProbe PRIVATE /W4 /WX /permissive-)
set_property(TARGET StreamlineProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME streamline-probe-self-test COMMAND StreamlineProbe --self-test)
set_tests_properties(streamline-probe-self-test PROPERTIES TIMEOUT 20)
